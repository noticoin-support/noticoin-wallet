cmake_minimum_required(VERSION 3.24)
project(NoticoinWallet VERSION 1.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
set(NOTICOIN_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(WEBVIEW2_INCLUDE "${NOTICOIN_ROOT}/desktop/vendor/webview2/build/native/include" CACHE PATH "WebView2 SDK headers")
set(RAPIDJSON_INCLUDE "${NOTICOIN_ROOT}/chain/external/rapidjson/include")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/webview2/WebView2.h")
    set(WEBVIEW2_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/webview2")
    set(RAPIDJSON_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/rapidjson")
endif()
qt_add_executable(Noticoin-Wallet WIN32 main.cpp wallet_service.cpp map_storage.cpp map_view.cpp wallet_window.cpp wallet_qr.cpp wallet_ui.cpp
    vendor/qrcodegen/qrcodegen.cpp)
qt_add_resources(Noticoin-Wallet "branding" PREFIX "/brand" FILES noticoin.png)
target_compile_definitions(Noticoin-Wallet PRIVATE NOTICOIN_WALLET_VERSION="${PROJECT_VERSION}")
target_include_directories(Noticoin-Wallet PRIVATE "${RAPIDJSON_INCLUDE}" "${NOTICOIN_ROOT}/chain")
target_link_libraries(Noticoin-Wallet PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network OpenSSL::Crypto)
if(WIN32)
    enable_language(RC)
    target_sources(Noticoin-Wallet PRIVATE wallet.rc)
    target_compile_definitions(Noticoin-Wallet PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
    target_include_directories(Noticoin-Wallet PRIVATE "${WEBVIEW2_INCLUDE}")
    target_link_libraries(Noticoin-Wallet PRIVATE ole32 dwmapi crypt32 winhttp)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(Qt6 REQUIRED COMPONENTS WebChannel WebEngineWidgets)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SecretService REQUIRED IMPORTED_TARGET libsecret-1)
    set_target_properties(Noticoin-Wallet PROPERTIES AUTOMOC ON)
    target_sources(Noticoin-Wallet PRIVATE map_webengine.cpp)
    target_link_libraries(Noticoin-Wallet PRIVATE Qt6::WebChannel Qt6::WebEngineWidgets PkgConfig::SecretService)
else()
    message(FATAL_ERROR "Noticoin Wallet supports Windows and Linux")
endif()
if(MINGW)
    target_compile_options(Noticoin-Wallet PRIVATE -Wall -Wextra -Wno-deprecated-declarations -finput-charset=UTF-8)
endif()
